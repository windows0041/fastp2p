cmake_minimum_required(VERSION 3.18)
project(fastp2p_c VERSION 2.0.5 LANGUAGES C)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/include/fastp2p.h" FASTP2P_HEADER_VERSION
     REGEX "^#define FASTP2P_VERSION ")
if(NOT FASTP2P_HEADER_VERSION STREQUAL "#define FASTP2P_VERSION \"${PROJECT_VERSION}\"")
  message(FATAL_ERROR "CMake and include/fastp2p.h must declare the same client version")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
# Pinned dependency headers must precede ambient Homebrew/CPATH headers.
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
include(CTest)
include(cmake/Dependencies.cmake)
option(FASTP2P_BUILD_SHARED "Build the public shared C library" ON)
option(FASTP2P_BUILD_CLI "Build command-line client" ON)
option(FASTP2P_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
set(FASTP2P_BOOTSTRAP_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../common/bootstrap.json" CACHE FILEPATH "Public gateway identity for this distribution")
if(FASTP2P_BUILD_CLI OR ANDROID)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${FASTP2P_BOOTSTRAP_FILE}")
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-bootstrap.py
    --input "${FASTP2P_BOOTSTRAP_FILE}" --output "${CMAKE_CURRENT_BINARY_DIR}/bootstrap_profile.h"
    RESULT_VARIABLE bootstrap_status)
  if(NOT bootstrap_status EQUAL 0)
    message(FATAL_ERROR "Invalid public gateway preset")
  endif()
  add_library(fastp2p_bootstrap STATIC ../common/bootstrap.c)
  target_include_directories(fastp2p_bootstrap PUBLIC ../common PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  set_target_properties(fastp2p_bootstrap PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
endif()
add_compile_options(-Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
add_compile_definitions(_GNU_SOURCE)
if(MINGW)
  add_link_options(-static-libgcc)
endif()
if(FASTP2P_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(fastp2p_platform STATIC src/platform.c)
target_include_directories(fastp2p_platform PUBLIC src)
set_target_properties(fastp2p_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
set_target_properties(fastp2p_platform PROPERTIES C_VISIBILITY_PRESET hidden)
if(WIN32)
  target_compile_definitions(fastp2p_platform PUBLIC _WIN32_WINNT=0x0A00)
  target_link_libraries(fastp2p_platform PUBLIC ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(fastp2p_platform PUBLIC Threads::Threads)
endif()
add_library(fastp2p_quic STATIC src/quic.c)
target_link_libraries(fastp2p_quic PUBLIC ${FASTP2P_QUIC_TARGET} fastp2p_platform)
target_include_directories(fastp2p_quic PUBLIC src)
set_target_properties(fastp2p_quic PROPERTIES POSITION_INDEPENDENT_CODE ON)
set_target_properties(fastp2p_quic PROPERTIES C_VISIBILITY_PRESET hidden)
if(FASTP2P_BUILD_SHARED)
  add_library(fastp2p SHARED src/client.c src/wire.c src/config.c src/socks.c src/secure.c src/update.c)
  target_compile_definitions(fastp2p PRIVATE FASTP2P_BUILDING_LIBRARY)
else()
  add_library(fastp2p STATIC src/client.c src/wire.c src/config.c src/socks.c src/secure.c src/update.c)
  target_compile_definitions(fastp2p PUBLIC FASTP2P_STATIC)
endif()
target_include_directories(fastp2p PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(fastp2p PRIVATE fastp2p_quic ${FASTP2P_WEB_TARGET} ${FASTP2P_JSON_TARGET} ${FASTP2P_CRYPTO_TARGET})
set_target_properties(fastp2p PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
set_target_properties(fastp2p PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
if(UNIX AND NOT APPLE AND FASTP2P_BUILD_SHARED)
  target_link_options(fastp2p PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/fastp2p.map")
endif()
if(APPLE AND FASTP2P_BUILD_SHARED)
  target_link_options(fastp2p PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/fastp2p.exports")
endif()
if(WIN32)
  set_target_properties(fastp2p PROPERTIES PREFIX "")
endif()
if(FASTP2P_BUILD_CLI)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-notices.py
    --prefix "${FASTP2P_DEPS_ROOT}" --output "${CMAKE_CURRENT_BINARY_DIR}/notices.h"
    RESULT_VARIABLE notices_status)
  if(NOT notices_status EQUAL 0)
    message(FATAL_ERROR "Cannot embed required distribution notices")
  endif()
  add_library(fastp2p_cli_core STATIC src/client.c src/wire.c src/config.c src/socks.c src/secure.c src/update.c ../common/enrollment.c)
  target_compile_definitions(fastp2p_cli_core PUBLIC FASTP2P_STATIC)
  target_include_directories(fastp2p_cli_core PUBLIC include src ../common)
  target_link_libraries(fastp2p_cli_core PUBLIC fastp2p_quic ${FASTP2P_WEB_TARGET} ${FASTP2P_JSON_TARGET} ${FASTP2P_CRYPTO_TARGET})
  add_executable(fastp2p-client src/main.c src/config_sync.c ../common/config_merge.c src/identity_file.c src/install.c src/autostart.c src/service.c src/installer_ui.c)
  target_include_directories(fastp2p-client PRIVATE ${CMAKE_CURRENT_BINARY_DIR})
  set_target_properties(fastp2p-client PROPERTIES OUTPUT_NAME fastp2p)
  target_link_libraries(fastp2p-client PRIVATE fastp2p_cli_core fastp2p_bootstrap)
  if(WIN32)
    target_link_libraries(fastp2p-client PRIVATE crypt32 advapi32 shell32 user32)
  endif()
  if(WIN32)
    enable_language(RC)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/fastp2p-client.rc" "1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/platform/windows/app.manifest\"\n")
    target_sources(fastp2p-client PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/fastp2p-client.rc")
  endif()
endif()
include(GNUInstallDirs)
if(FASTP2P_BUILD_CLI)
  if(APPLE)
    set_target_properties(fastp2p-client PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  elseif(UNIX)
    set_target_properties(fastp2p-client PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  install(TARGETS fastp2p-client RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS fastp2p EXPORT fastp2pTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/fastp2p.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(FASTP2P_BUILD_SHARED)
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/fastp2pConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/fastp2pConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fastp2p)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/fastp2pConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(EXPORT fastp2pTargets NAMESPACE fastp2p:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fastp2p)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fastp2pConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/fastp2pConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fastp2p)
endif()
add_executable(example-embed EXCLUDE_FROM_ALL examples/embed.c)
target_link_libraries(example-embed PRIVATE fastp2p)
if(ANDROID)
  target_link_options(fastp2p PRIVATE "LINKER:-z,max-page-size=16384")
  add_library(fastp2p_jni SHARED android/jni/client_jni.c)
  target_link_libraries(fastp2p_jni PRIVATE fastp2p fastp2p_bootstrap)
  target_link_options(fastp2p_jni PRIVATE "LINKER:-z,max-page-size=16384")
  set_target_properties(fastp2p_jni PROPERTIES C_VISIBILITY_PRESET hidden)
endif()

  add_test(NAME platform COMMAND test-platform)
  add_executable(quic-interop tests/quic_interop.c)
  target_link_libraries(quic-interop PRIVATE fastp2p_quic)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(FASTP2P_BUILD_CLI)
    if(WIN32)
      add_executable(test-update-windows tests/test_update_windows.c)

      target_compile_options(test-service-windows PRIVATE -UNDEBUG)

      target_link_libraries(test-profile-probe PRIVATE fastp2p_cli_core fastp2p_bootstrap crypt32 advapi32 shell32 user32)
      # Requires an elevated isolated runner; the local CTest suite need not be admin.

    add_executable(test-session-binding tests/test_session_binding.c)
    target_link_libraries(test-session-binding PRIVATE fastp2p_cli_core)

  add_executable(test-nat-classify tests/test_nat_classify.c)
  target_link_libraries(test-nat-classify PRIVATE fastp2p_cli_core)
  target_compile_options(test-nat-classify PRIVATE -UNDEBUG)
  add_test(NAME nat-classify COMMAND test-nat-classify)
    add_test(NAME cli-identity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/identity_cli.py
      --client $<TARGET_FILE:fastp2p-client>)
    set_tests_properties(cli-identity PROPERTIES TIMEOUT 45)
  endif()
  add_executable(test-wire tests/test_wire.c src/wire.c)
  target_include_directories(test-wire PRIVATE src)
target_link_libraries(test-wire PRIVATE ${FASTP2P_JSON_TARGET} ${FASTP2P_CRYPTO_TARGET})
  add_executable(test-secure tests/test_secure.c src/secure.c)
  target_include_directories(test-secure PRIVATE src)
  target_link_libraries(test-secure PRIVATE ${FASTP2P_CRYPTO_TARGET})
  target_compile_options(test-secure PRIVATE -UNDEBUG)
  add_test(NAME secure COMMAND test-secure)
  add_executable(test-api tests/test_api.c)
  target_link_libraries(test-api PRIVATE fastp2p)
  target_compile_options(test-api PRIVATE -UNDEBUG)
  target_compile_options(test-wire PRIVATE -UNDEBUG)
  add_test(NAME wire COMMAND test-wire)

  add_test(NAME tls-validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_negative.py
    --client $<TARGET_FILE:fastp2p-client> --cert-script ${CMAKE_CURRENT_SOURCE_DIR}/../../scripts/tls/generate-cert.sh)
  set_tests_properties(tls-validation PROPERTIES TIMEOUT 45)
