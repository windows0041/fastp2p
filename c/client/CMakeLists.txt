cmake_minimum_required(VERSION 3.18)
project(openp2p_c VERSION 0.4.0 LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
# Pinned dependency headers must precede ambient Homebrew/CPATH headers.
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
include(CTest)
include(cmake/Dependencies.cmake)
option(OPENP2P_BUILD_SHARED "Build the public shared C library" ON)
option(OPENP2P_BUILD_CLI "Build command-line client" ON)
option(OPENP2P_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
add_compile_options(-Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
add_compile_definitions(_GNU_SOURCE)
if(MINGW)
  add_link_options(-static-libgcc)
endif()
if(OPENP2P_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(op2p_platform STATIC src/platform.c)
target_include_directories(op2p_platform PUBLIC src)
set_target_properties(op2p_platform PROPERTIES POSITION_INDEPENDENT_CODE ON)
set_target_properties(op2p_platform PROPERTIES C_VISIBILITY_PRESET hidden)
if(WIN32)
  target_compile_definitions(op2p_platform PUBLIC _WIN32_WINNT=0x0A00)
  target_link_libraries(op2p_platform PUBLIC ws2_32)
else()
  find_package(Threads REQUIRED)
  target_link_libraries(op2p_platform PUBLIC Threads::Threads)
endif()
add_library(op2p_quic STATIC src/quic.c)
target_link_libraries(op2p_quic PUBLIC ${OP2P_QUIC_TARGET} op2p_platform)
target_include_directories(op2p_quic PUBLIC src)
set_target_properties(op2p_quic PROPERTIES POSITION_INDEPENDENT_CODE ON)
set_target_properties(op2p_quic PROPERTIES C_VISIBILITY_PRESET hidden)
if(OPENP2P_BUILD_SHARED)
  add_library(openp2p SHARED src/client.c src/wire.c src/config.c src/socks.c)
  target_compile_definitions(openp2p PRIVATE OP2P_BUILDING_LIBRARY)
else()
  add_library(openp2p STATIC src/client.c src/wire.c src/config.c src/socks.c)
  target_compile_definitions(openp2p PUBLIC OP2P_STATIC)
endif()
target_include_directories(openp2p PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
target_link_libraries(openp2p PRIVATE op2p_quic ${OP2P_WEB_TARGET} ${OP2P_JSON_TARGET} ${OP2P_CRYPTO_TARGET})
set_target_properties(openp2p PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 1)
set_target_properties(openp2p PROPERTIES POSITION_INDEPENDENT_CODE ON C_VISIBILITY_PRESET hidden)
if(UNIX AND NOT APPLE AND OPENP2P_BUILD_SHARED)
  target_link_options(openp2p PRIVATE "LINKER:--version-script=${CMAKE_CURRENT_SOURCE_DIR}/src/openp2p.map")
endif()
if(APPLE AND OPENP2P_BUILD_SHARED)
  target_link_options(openp2p PRIVATE "LINKER:-exported_symbols_list,${CMAKE_CURRENT_SOURCE_DIR}/src/openp2p.exports")
endif()
if(WIN32)
  set_target_properties(openp2p PROPERTIES PREFIX "")
endif()
if(OPENP2P_BUILD_CLI)
  add_executable(openp2p-c src/main.c src/identity_file.c)
  target_link_libraries(openp2p-c PRIVATE openp2p ${OP2P_JSON_TARGET})
  if(WIN32)
    enable_language(RC)
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/openp2p-client.rc" "1 24 \"${CMAKE_CURRENT_SOURCE_DIR}/platform/windows/app.manifest\"\n")
    target_sources(openp2p-c PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/openp2p-client.rc")
  endif()
endif()
include(GNUInstallDirs)
if(OPENP2P_BUILD_CLI)
  if(APPLE)
    set_target_properties(openp2p-c PROPERTIES INSTALL_RPATH "@loader_path/../${CMAKE_INSTALL_LIBDIR}")
  elseif(UNIX)
    set_target_properties(openp2p-c PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
  endif()
  install(TARGETS openp2p-c RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
install(TARGETS openp2p EXPORT openp2pTargets LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/openp2p.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
if(OPENP2P_BUILD_SHARED)
  include(CMakePackageConfigHelpers)
  configure_package_config_file(cmake/openp2pConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/openp2pConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/openp2p)
  write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/openp2pConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
  install(EXPORT openp2pTargets NAMESPACE openp2p:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/openp2p)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/openp2pConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/openp2pConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/openp2p)
endif()
add_executable(example-embed EXCLUDE_FROM_ALL examples/embed.c)
target_link_libraries(example-embed PRIVATE openp2p)
if(ANDROID)
  target_link_options(openp2p PRIVATE "LINKER:-z,max-page-size=16384")
  add_library(openp2p_jni SHARED android/jni/client_jni.c)
  target_link_libraries(openp2p_jni PRIVATE openp2p)
  target_link_options(openp2p_jni PRIVATE "LINKER:-z,max-page-size=16384")
  set_target_properties(openp2p_jni PROPERTIES C_VISIBILITY_PRESET hidden)
endif()
if(BUILD_TESTING)
  add_executable(test-socks tests/test_socks.c src/socks.c)
  target_include_directories(test-socks PRIVATE src)
  target_link_libraries(test-socks PRIVATE ${OP2P_CRYPTO_TARGET})
  target_compile_options(test-socks PRIVATE -UNDEBUG)
  add_test(NAME socks-parser COMMAND test-socks)
  add_executable(test-platform tests/test_platform.c)
  target_link_libraries(test-platform PRIVATE op2p_platform)
  target_compile_options(test-platform PRIVATE -UNDEBUG)
  add_test(NAME platform COMMAND test-platform)
  add_executable(quic-interop tests/quic_interop.c)
  target_link_libraries(quic-interop PRIVATE op2p_quic)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  if(OPENP2P_BUILD_CLI)
    add_test(NAME cli-identity COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/identity_cli.py
      --client $<TARGET_FILE:openp2p-c>)
    set_tests_properties(cli-identity PROPERTIES TIMEOUT 45)
  endif()
  add_executable(test-wire tests/test_wire.c src/wire.c)
  target_include_directories(test-wire PRIVATE src)
target_link_libraries(test-wire PRIVATE ${OP2P_JSON_TARGET} ${OP2P_CRYPTO_TARGET})
  add_executable(test-api tests/test_api.c)
  target_link_libraries(test-api PRIVATE openp2p)
  target_compile_options(test-api PRIVATE -UNDEBUG)
  target_compile_options(test-wire PRIVATE -UNDEBUG)
  add_test(NAME wire COMMAND test-wire)
  add_test(NAME api COMMAND test-api)
  add_test(NAME tls-validation COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/tls_negative.py
    --client $<TARGET_FILE:openp2p-c> --cert-script ${CMAKE_CURRENT_SOURCE_DIR}/../../go/gateway/scripts/generate-cert.sh)
  set_tests_properties(tls-validation PROPERTIES TIMEOUT 45)
endif()
