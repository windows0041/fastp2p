cmake_minimum_required(VERSION 3.18)
project(openp2p_sdk_consumer LANGUAGES C)
set(CMAKE_C_STANDARD 11)
find_package(openp2p 0.2 CONFIG REQUIRED)
add_executable(sdk-consumer ../test_api.c)
target_link_libraries(sdk-consumer PRIVATE openp2p::openp2p)
if(MSVC)
  target_compile_options(sdk-consumer PRIVATE /UNDEBUG)
else()
  target_compile_options(sdk-consumer PRIVATE -UNDEBUG)
endif()
