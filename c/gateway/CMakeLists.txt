cmake_minimum_required(VERSION 3.18)
project(openp2p_c_gateway VERSION 0.1.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The gateway targets Linux; use WSL on Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
include(CTest)
option(OPENP2P_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LWS REQUIRED IMPORTED_TARGET libwebsockets>=4.1)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c>=0.16)
find_package(OpenSSL 3.0 REQUIRED)
add_compile_options(-Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
add_compile_definitions(_GNU_SOURCE)
if(OPENP2P_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(gateway_protocol STATIC src/protocol.c src/security.c)
target_include_directories(gateway_protocol PUBLIC src)
target_link_libraries(gateway_protocol PUBLIC PkgConfig::JSON OpenSSL::Crypto)
add_executable(openp2p-gateway-c src/gateway.c src/nat.c)
target_link_libraries(openp2p-gateway-c PRIVATE gateway_protocol PkgConfig::LWS OpenSSL::SSL)
include(GNUInstallDirs)
install(TARGETS openp2p-gateway-c RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING)
  add_executable(test-gateway-protocol tests/test_protocol.c)
  target_link_libraries(test-gateway-protocol PRIVATE gateway_protocol)
  target_compile_options(test-gateway-protocol PRIVATE -UNDEBUG)
  add_test(NAME gateway-protocol COMMAND test-gateway-protocol)
endif()
