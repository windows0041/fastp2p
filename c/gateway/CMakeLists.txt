cmake_minimum_required(VERSION 3.18)
project(fastp2p_c_gateway VERSION 1.0.0 LANGUAGES C)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "The gateway targets Linux; use WSL on Windows")
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_NO_SYSTEM_FROM_IMPORTED ON)
include(CTest)
option(FASTP2P_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LWS REQUIRED IMPORTED_TARGET libwebsockets>=4.1)
pkg_check_modules(JSON REQUIRED IMPORTED_TARGET json-c>=0.16)
find_package(OpenSSL 3.0 REQUIRED)
find_package(SQLite3 3.24 REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(WEB_HEADER "${CMAKE_CURRENT_BINARY_DIR}/web_assets.h")
add_custom_command(OUTPUT "${WEB_HEADER}"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/scripts/embed-web.py"
          "${CMAKE_CURRENT_SOURCE_DIR}/web" "${WEB_HEADER}"
  DEPENDS scripts/embed-web.py web/index.html web/app.js web/style.css web/favicon.svg
  VERBATIM)
add_compile_options(-Wall -Wextra -Wpedantic -Werror=implicit-function-declaration)
add_compile_definitions(_GNU_SOURCE)
if(FASTP2P_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
add_library(gateway_protocol STATIC src/protocol.c src/security.c ../common/enrollment.c)
target_include_directories(gateway_protocol PUBLIC src ../common)
target_link_libraries(gateway_protocol PUBLIC PkgConfig::JSON OpenSSL::Crypto)
add_library(gateway_store STATIC src/store.c)
target_link_libraries(gateway_store PUBLIC gateway_protocol SQLite::SQLite3 PkgConfig::LWS)
add_executable(fastp2p-gateway src/gateway.c src/nat.c "${WEB_HEADER}")
target_include_directories(fastp2p-gateway PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(fastp2p-gateway PRIVATE gateway_store PkgConfig::LWS OpenSSL::SSL)
include(GNUInstallDirs)
install(TARGETS fastp2p-gateway RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
if(BUILD_TESTING)
  add_executable(test-gateway-store tests/test_store.c)
  target_link_libraries(test-gateway-store PRIVATE gateway_store)
  target_compile_options(test-gateway-store PRIVATE -UNDEBUG)
  add_test(NAME gateway-store COMMAND test-gateway-store)
  add_executable(test-gateway-protocol tests/test_protocol.c)
  target_link_libraries(test-gateway-protocol PRIVATE gateway_protocol)
  target_compile_options(test-gateway-protocol PRIVATE -UNDEBUG)
  add_test(NAME gateway-protocol COMMAND test-gateway-protocol)
endif()
